"""同花顺自动下单客户端(宿主机侧)。

对接 Windows 虚拟机内运行的 thsauto HTTP 服务。

设计前提 —— 服务端是 GUI 自动化 + 屏幕 OCR，**返回值不可信**:
  * 下单接口可能返回 code=2 "获取结果失败,请自行确认订单状态";
  * OCR 漏识别会触发服务端补发回车,存在重复下单风险。
因此本客户端的策略是:
  1) 下单前做风控检查(次数/金额/白名单)并落盘,防止进程重启后绕过熔断;
  2) 下单前抓一次委托快照,下单后回查委托列表做差集,以**回查结果**为准;
  3) 任何无法确认的状态一律标记 UNKNOWN 并停止后续自动下单,交人工处理。
"""
from __future__ import annotations

import json
import logging
import os
import threading
from dataclasses import dataclass
from datetime import date
from enum import Enum
from typing import Any
from urllib.parse import urlencode

import requests

from .config import Config


class OrderStatus(str, Enum):
    CONFIRMED = "confirmed"   # 回查到委托,确认已报
    REJECTED = "rejected"     # 明确失败,未下单
    UNKNOWN = "unknown"       # 无法确认,必须人工介入
    BLOCKED = "blocked"       # 被本地风控拦截,未发往服务端
    DRY_RUN = "dry_run"       # 空跑模式,未真实下单


@dataclass
class OrderResult:
    status: OrderStatus
    side: str
    stock_no: str
    price: float | None
    amount: int
    entrust_no: str | None = None
    raw: dict[str, Any] | None = None
    msg: str = ""

    @property
    def ok(self) -> bool:
        return self.status is OrderStatus.CONFIRMED

    @property
    def needs_attention(self) -> bool:
        return self.status is OrderStatus.UNKNOWN


class RiskBlocked(Exception):
    """本地风控拦截,未向服务端发出任何请求。"""


class ThsClient:
    def __init__(self, config: Config | None = None):
        self.cfg = config or Config()
        self.base = f"http://{self.cfg.host}:{self.cfg.port}/thsauto"
        self._lock = threading.Lock()
        self._halted = False
        self._halt_reason = ""
        self._log = self._setup_logger()
        self._state = self._load_state()

    # ------------------------------------------------------------------
    # 基础设施
    # ------------------------------------------------------------------
    def _setup_logger(self) -> logging.Logger:
        log = logging.getLogger("ths_auto")
        if log.handlers:
            return log
        log.setLevel(logging.INFO)
        os.makedirs(os.path.dirname(self.cfg.log_file) or ".", exist_ok=True)
        fmt = logging.Formatter("%(asctime)s [%(levelname)s] %(message)s")
        fh = logging.FileHandler(self.cfg.log_file, encoding="utf-8")
        fh.setFormatter(fmt)
        sh = logging.StreamHandler()
        sh.setFormatter(fmt)
        log.addHandler(fh)
        log.addHandler(sh)
        return log

    def _load_state(self) -> dict:
        """日内计数落盘,进程重启不会绕过熔断。"""
        today = date.today().isoformat()
        blank = {"date": today, "order_count": 0, "turnover": 0.0}
        try:
            with open(self.cfg.state_file, encoding="utf-8") as f:
                state = json.load(f)
            if state.get("date") != today:
                return blank
            return state
        except (FileNotFoundError, json.JSONDecodeError):
            return blank

    def _save_state(self) -> None:
        os.makedirs(os.path.dirname(self.cfg.state_file) or ".", exist_ok=True)
        with open(self.cfg.state_file, "w", encoding="utf-8") as f:
            json.dump(self._state, f, ensure_ascii=False, indent=2)

    def _get(self, path: str, **params) -> dict:
        url = f"{self.base}/{path}"
        if params:
            url = f"{url}?{urlencode(params)}"
        resp = requests.get(url, timeout=self.cfg.timeout)
        resp.raise_for_status()
        return resp.json()

    # ------------------------------------------------------------------
    # 熔断
    # ------------------------------------------------------------------
    def halt(self, reason: str) -> None:
        """触发熔断,后续下单一律拒绝,直到人工 resume()。"""
        self._halted = True
        self._halt_reason = reason
        self._log.error("!!! 已熔断: %s", reason)

    def resume(self) -> None:
        self._log.warning("人工解除熔断(原因曾为: %s)", self._halt_reason)
        self._halted = False
        self._halt_reason = ""

    @property
    def halted(self) -> bool:
        return self._halted

    def _check_risk(self, stock_no: str, price: float | None, amount: int) -> None:
        if self._halted:
            raise RiskBlocked(f"处于熔断状态: {self._halt_reason}")

        if amount <= 0 or amount % 100 != 0:
            raise RiskBlocked(f"数量非法(须为100整数倍): {amount}")

        if price is None and not self.cfg.allow_market_order:
            raise RiskBlocked("未指定价格且未开启市价单")

        if self.cfg.stock_whitelist and stock_no not in self.cfg.stock_whitelist:
            raise RiskBlocked(f"{stock_no} 不在白名单内")

        notional = (price or 0) * amount
        if price is not None and notional > self.cfg.max_amount_per_order:
            raise RiskBlocked(
                f"单笔金额 {notional:.0f} 超限 {self.cfg.max_amount_per_order:.0f}"
            )

        if self._state["order_count"] >= self.cfg.max_orders_per_day:
            raise RiskBlocked(
                f"日内下单次数已达上限 {self.cfg.max_orders_per_day}"
            )

        if self._state["turnover"] + notional > self.cfg.max_turnover_per_day:
            raise RiskBlocked(
                f"日内成交额将超上限 {self.cfg.max_turnover_per_day:.0f}"
            )

    # ------------------------------------------------------------------
    # 查询
    # ------------------------------------------------------------------
    def balance(self) -> dict:
        return self._get("balance")

    def position(self) -> dict:
        return self._get("position")

    def active_orders(self) -> Any:
        return self._get("orders/active")

    def filled_orders(self) -> Any:
        return self._get("orders/filled")

    def cancel(self, entrust_no: str) -> dict:
        self._log.info("撤单 entrust_no=%s", entrust_no)
        return self._get("cancel", entrust_no=entrust_no)

    # ------------------------------------------------------------------
    # 委托快照 / 回查
    # ------------------------------------------------------------------
    @staticmethod
    def _rows(payload: Any) -> list[dict]:
        """服务端表格返回形态不统一,这里归一成 list[dict]。"""
        if isinstance(payload, list):
            return [r for r in payload if isinstance(r, dict)]
        if isinstance(payload, dict):
            for key in ("data", "rows", "result", "list"):
                val = payload.get(key)
                if isinstance(val, list):
                    return [r for r in val if isinstance(r, dict)]
        return []

    @staticmethod
    def _entrust_ids(rows: list[dict]) -> set[str]:
        ids: set[str] = set()
        for row in rows:
            for key, val in row.items():
                if "编号" in str(key) or "entrust" in str(key).lower():
                    if val not in (None, ""):
                        ids.add(str(val).strip())
        return ids

    def _snapshot_ids(self) -> set[str]:
        """抓当前全部委托编号(未成+已成),用于下单后做差集。"""
        ids: set[str] = set()
        for fetch in (self.active_orders, self.filled_orders):
            try:
                ids |= self._entrust_ids(self._rows(fetch()))
            except Exception as exc:   # 查询失败不应阻断下单前流程
                self._log.warning("快照查询失败: %s", exc)
        return ids

    def _verify(
        self, before: set[str], stock_no: str, hinted_no: str | None
    ) -> tuple[OrderStatus, str | None, str]:
        """以回查结果判定真实状态,而非信任下单接口返回值。"""
        import time

        time.sleep(self.cfg.verify_delay)
        for attempt in range(1, self.cfg.verify_retries + 1):
            try:
                rows = self._rows(self.active_orders()) + self._rows(self.filled_orders())
            except Exception as exc:
                self._log.warning("回查第%d次失败: %s", attempt, exc)
                time.sleep(self.cfg.verify_interval)
                continue

            now_ids = self._entrust_ids(rows)
            new_ids = now_ids - before

            # 服务端 OCR 给出的编号若能在委托列表里找到,是最强证据
            if hinted_no and hinted_no in now_ids:
                return OrderStatus.CONFIRMED, hinted_no, "OCR编号已在委托列表中核实"

            # 只新增一笔,且该笔标的匹配 -> 确认
            matched = [
                r for r in rows
                if self._entrust_ids([r]) & new_ids
                and stock_no in json.dumps(r, ensure_ascii=False)
            ]
            if len(matched) == 1:
                eid = next(iter(self._entrust_ids(matched)), None)
                return OrderStatus.CONFIRMED, eid, "回查新增一笔匹配委托"

            if len(matched) > 1:
                # 出现多笔新增 —— 很可能是服务端补发回车造成重复下单
                return (
                    OrderStatus.UNKNOWN,
                    None,
                    f"回查发现 {len(matched)} 笔新增委托,疑似重复下单,立即人工核对!",
                )

            time.sleep(self.cfg.verify_interval)

        return OrderStatus.UNKNOWN, None, "回查未发现新增委托,无法确认是否已报单"

    # ------------------------------------------------------------------
    # 下单
    # ------------------------------------------------------------------
    def _place(
        self, side: str, stock_no: str, price: float | None, amount: int, kc: bool
    ) -> OrderResult:
        with self._lock:
            try:
                self._check_risk(stock_no, price, amount)
            except RiskBlocked as exc:
                self._log.warning("风控拦截 %s %s: %s", side, stock_no, exc)
                return OrderResult(
                    OrderStatus.BLOCKED, side, stock_no, price, amount, msg=str(exc)
                )

            if self.cfg.dry_run:
                self._log.info(
                    "[DRY-RUN] %s %s price=%s amount=%s (未真实下单)",
                    side, stock_no, price, amount,
                )
                return OrderResult(
                    OrderStatus.DRY_RUN, side, stock_no, price, amount,
                    msg="dry_run=True,未发送真实订单",
                )

            before = self._snapshot_ids()

            params: dict[str, Any] = {"stock_no": stock_no, "amount": amount}
            if price is not None:
                params["price"] = f"{price:.3f}"
            path = f"{side}/kc" if kc else side

            self._log.info("发送 %s %s price=%s amount=%s", side, stock_no, price, amount)
            try:
                raw = self._get(path, **params)
            except Exception as exc:
                # 请求异常不代表没下单(可能超时但服务端已执行),必须回查
                self._log.error("下单请求异常: %s —— 转入回查确认", exc)
                status, eid, msg = self._verify(before, stock_no, None)
                if status is OrderStatus.UNKNOWN:
                    self.halt(f"{side} {stock_no} 请求异常且无法确认状态")
                return OrderResult(
                    status, side, stock_no, price, amount, eid,
                    msg=f"请求异常({exc}); 回查: {msg}",
                )

            hinted = raw.get("entrust_no") if isinstance(raw, dict) else None
            status, eid, msg = self._verify(before, stock_no, hinted)

            # 计数以"确实发出过请求"为准,宁可多记不可漏记
            self._state["order_count"] += 1
            self._state["turnover"] += (price or 0) * amount
            self._save_state()

            if status is OrderStatus.UNKNOWN:
                self.halt(f"{side} {stock_no} 状态未知: {msg}")

            self._log.info("结果 %s %s -> %s (%s)", side, stock_no, status.value, msg)
            return OrderResult(status, side, stock_no, price, amount, eid, raw, msg)

    def buy(self, stock_no: str, price: float | None, amount: int, kc: bool = False):
        return self._place("buy", stock_no, price, amount, kc)

    def sell(self, stock_no: str, price: float | None, amount: int, kc: bool = False):
        return self._place("sell", stock_no, price, amount, kc)

    def ping(self) -> bool:
        """连通性检查:能取到资金账户即视为服务可用。"""
        try:
            self.balance()
            return True
        except Exception as exc:
            self._log.error("连通失败: %s", exc)
            return False
